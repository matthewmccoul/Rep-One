cmake_minimum_required(VERSION 3.16)
project(MordorCore CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# The rules core: no I/O, no exceptions, no RTTI, no globals (so it can move into an Unreal module unchanged).
file(GLOB CORE_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/*.cpp)
add_library(mordor_core STATIC ${CORE_SRC})
target_include_directories(mordor_core PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_compile_options(mordor_core PRIVATE -Wall -Wextra -fno-exceptions -fno-rtti)

# Tests: unit tests derived from the SRD, game-flow tests, and (separately) acceptance tests translated from the reference implementation's own assertions.
file(GLOB TEST_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/*.cpp)
set(ENGINE_TEST_SRC ${TEST_SRC})
list(FILTER ENGINE_TEST_SRC INCLUDE REGEX "test_engine_|test_main")
list(FILTER TEST_SRC EXCLUDE REGEX "test_engine_")
add_executable(mordor_tests ${TEST_SRC})
target_link_libraries(mordor_tests PRIVATE mordor_core)
target_compile_definitions(mordor_tests PRIVATE MORDOR_DATA_FILE="${CMAKE_SOURCE_DIR}/data/combat-data.json")
target_compile_options(mordor_tests PRIVATE -Wall -Wextra)
add_executable(mordor_engine_tests ${ENGINE_TEST_SRC})
target_link_libraries(mordor_engine_tests PRIVATE mordor_core)
target_compile_definitions(mordor_engine_tests PRIVATE MORDOR_DATA_FILE="${CMAKE_SOURCE_DIR}/data/combat-data.json")
target_compile_options(mordor_engine_tests PRIVATE -Wall -Wextra)
enable_testing()
add_test(NAME srd_and_flow COMMAND mordor_tests)
add_test(NAME engine_acceptance COMMAND mordor_engine_tests)

# Console front end (the demo); lives outside the core and talks to it only through the public API.
file(GLOB APP_SRC CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/apps/console/*.cpp)
if(APP_SRC)
  add_executable(mordor_console ${APP_SRC})
  target_link_libraries(mordor_console PRIVATE mordor_core)
  target_compile_definitions(mordor_console PRIVATE MORDOR_DATA_FILE="${CMAKE_SOURCE_DIR}/data/combat-data.json")
endif()
