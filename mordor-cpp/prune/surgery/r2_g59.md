// Reference HexDigit: value of a hex digit, or -1.
i32 hexDigit(char c) {
    if (c >= '0' && c <= '9') return static_cast<i32>(c - '0');
    if (c >= 'A' && c <= 'F') return static_cast<i32>(c - 'A') + 10;
    if (c >= 'a' && c <= 'f') return static_cast<i32>(c - 'a') + 10;
    return -1;
}
