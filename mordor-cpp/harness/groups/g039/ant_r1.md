using System;
using System.Collections.Generic;
using System.Reflection;
using System.Text;
using System.Text.Json;
using Mordor.Core;

public static class Case
{
    public static string Run(JsonElement a, Random rng)
    {
        int n = Get(a, "n");
        var list = new List<Character>();
        for (int i = 0; i < n; i++)
        {
            var c = Character.Create(((char)('A' + i)).ToString(), RaceData.ByName("Human"), Alignment.Neutral, Sex.Male, default);
            c.CurHits = Get(a, "h" + i);
            c.DungeonLevel = Get(a, "l" + i);
            c.X = Get(a, "x" + i);
            c.Y = Get(a, "y" + i);
            list.Add(c);
        }
        var party = new Party(list.ToArray());
        // SelectedIndex has a private setter; set it through the existing setter.
        typeof(Party).GetProperty("SelectedIndex")!.GetSetMethod(true)!.Invoke(party, new object[] { Get(a, "sel") });

        party.RemoveTheDead(); // returned messages and rng are not part of the canonical string

        return Dump(party);
    }

    static int Get(JsonElement a, string key) => a.TryGetProperty(key, out var v) ? v.GetInt32() : 0;

    static string Dump(Party p)
    {
        var sb = new StringBuilder();
        sb.Append(p.Size).Append('|').Append(p.SelectedIndex);
        foreach (var c in p.Members)
        {
            sb.Append('|').Append(c.Name).Append('|').Append(c.CurHits)
              .Append('|').Append(c.IsDead ? 1 : 0)
              .Append('|').Append(c.Stoned ? 1 : 0)
              .Append('|').Append(c.DungeonLevel)
              .Append('|').Append(c.X).Append('|').Append(c.Y);
        }
        return sb.ToString();
    }
}

#include "case_api.h"
#include "mordor/state_character.h"

void mordor_export_partyb_prune(Ctx& ctx, Party& party);

static i32 slot(const JsonValue& a, const char* prefix, i32 i) {
    Str k = Str(prefix) + I(i);
    return static_cast<i32>(A(a, k.c_str()).asInt());
}

static void fillParty(Party& p, const JsonValue& a) {
    const i32 n = static_cast<i32>(A(a, "n").asInt());
    for (i32 i = 0; i < n; ++i) {
        Character& m = p.members[i];
        m.name = Str(1, static_cast<char>('A' + i));
        m.vitals.curHits = slot(a, "h", i);
        m.location.dungeonLevel = slot(a, "l", i);
        m.location.x = slot(a, "x", i);
        m.location.y = slot(a, "y", i);
    }
    p.size = n;
    p.selectedIndex = static_cast<i32>(A(a, "sel").asInt());
}

static Str dump(Party& p) {
    Str s = I(p.size) + "|" + I(p.selectedIndex);
    for (i32 i = 0; i < p.size; ++i) {
        Character& m = p.members[i];
        s += "|" + m.name + "|" + I(m.vitals.curHits)
           + "|" + I(m.isDead() ? 1 : 0)
           + "|" + I(m.deathState == DeathState::Stoned ? 1 : 0)
           + "|" + I(m.location.dungeonLevel)
           + "|" + I(m.location.x) + "|" + I(m.location.y);
    }
    return s;
}

static Str run_pruneTheDead_partyb(const JsonValue& a, Ctx& ctx) {
    Party p;
    fillParty(p, a);
    mordor_export_partyb_prune(ctx, p);
    return dump(p);
}

static Str run_pruneTheDead_death1a(const JsonValue& a, Ctx& ctx) {
    Party p;
    fillParty(p, a);
    Arr<bool, kPartyMaxSize> counted{};
    mordor_export_death1a_prune(ctx, p, counted);
    return dump(p);
}

Notes on coverage and gaps:

- **Messages are not compared.** The C# method returns the "has died!" messages, but the C++ event sink interface (`IEventSink`) is not shown, so I could not capture `ctx.say` output. Neither side includes messages in its string.
- **Dead and stoned casualties are not exercised.** The shown C# `Character` API has no setter for `IsDead` or `Stoned`, so every case makes a casualty by setting `CurHits` to 0 or below. The `DeathState::Stoned` branch and the `setDead` branch are not covered.
- **Removed members are not compared.** Their final hits and death state are checked on the C# side only through callbacks I cannot compare, and on the C++ side they are gone from the array after removal. The output covers only the members left in the party.
- **`countedDead` has no reference counterpart.** The C# method takes no such argument, so the death1a copy always receives an all-false array. Its counting behaviour is not tested.
- **`SelectedIndex` is set by reflection** on its existing private setter, because the shown C# API has no public setter.
- **`Character.Create` is passed `default` for the stats argument**, since its parameter type is not shown. If `Create` rejects that value, the C# side will throw and the cases will need a real stats value.
- **Expected divergences the test should expose:**
  - `partyb` clamps `selectedIndex` only when a member is removed, while the reference always clamps (cases 19 and 21 target this).
  - `death1a` never clamps, never zeroes hits, and never moves survivors to the leader's square.
  - The `size <= 1` guard returns early in both the reference and `partyb`, so case 2 should still match.
